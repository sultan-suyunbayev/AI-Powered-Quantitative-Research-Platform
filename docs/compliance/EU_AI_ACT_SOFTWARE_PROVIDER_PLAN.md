```python
"""
Article 50 EU AI Act - Transparency Disclosure System.

References:
- Article 50(1): AI interaction disclosure
- Article 50(4): Synthetic content marking
- Recital 132: Transparency for limited-risk AI
"""
from enum import Enum
from dataclasses import dataclass, field
from datetime import datetime
from typing import Optional
import hashlib
import json

class DisclosureType(Enum):
    """Article 50 disclosure categories."""
    AI_INTERACTION = "ai_interaction"      # Art. 50(1)
    SYNTHETIC_CONTENT = "synthetic_content" # Art. 50(2)
    EMOTION_RECOGNITION = "emotion_recognition" # Art. 50(3)
    DEEP_FAKE = "deep_fake"                # Art. 50(4)

class DisclosureContext(Enum):
    """Context where disclosure is shown."""
    REGISTRATION = "registration"
    STRATEGY_CREATION = "strategy_creation"
    LIVE_TRADING_ACTIVATION = "live_trading_activation"
    MODEL_OUTPUT = "model_output"
    API_RESPONSE = "api_response"

@dataclass
class AIDisclosure:
    """
    Standard AI disclosure per Article 50.

    Must inform user that:
    1. They are interacting with an AI system
    2. The system uses machine learning (RL/PPO)
    3. Outputs are AI-generated predictions
    """
    disclosure_id: str
    disclosure_type: DisclosureType
    context: DisclosureContext
    timestamp: datetime
    user_id: str
    acknowledged: bool = False
    acknowledgment_timestamp: Optional[datetime] = None

    # Required content per Article 50
    ai_system_name: str = "Distributional PPO Trading Model"
    ai_system_type: str = "Reinforcement Learning"
    ai_purpose: str = "Trading signal generation"

    def generate_disclosure_text(self, language: str = "en") -> str:
        """Generate Article 50 compliant disclosure text."""
        texts = {
            "en": {
                "header": "AI SYSTEM DISCLOSURE",
                "body": (
                    "This service uses an Artificial Intelligence (AI) system "
                    "based on machine learning technology (Reinforcement Learning). "
                    "Trading signals and predictions are generated by the AI model, "
                    "not by human analysts. The AI system has limitations and may "
                    "produce inaccurate predictions. You retain full responsibility "
                    "for all trading decisions."
                ),
                "acknowledge": "I understand that I am interacting with an AI system"
            },
            "ru": {
                "header": "РАСКРЫТИЕ ИНФОРМАЦИИ ОБ ИИ-СИСТЕМЕ",
                "body": (
                    "Данный сервис использует систему искусственного интеллекта (ИИ) "
                    "на основе технологии машинного обучения (Reinforcement Learning). "
                    "Торговые сигналы и прогнозы генерируются ИИ-моделью, "
                    "а не аналитиками. ИИ-система имеет ограничения и может "
                    "давать неточные прогнозы. Вы несёте полную ответственность "
                    "за все торговые решения."
                ),
                "acknowledge": "Я понимаю, что взаимодействую с ИИ-системой"
            }
        }
        return texts.get(language, texts["en"])

@dataclass
class DisclosureRequirement:
    """Defines when disclosure is required per Article 50."""
    context: DisclosureContext
    required: bool
    reason: str
    article_reference: str

DISCLOSURE_REQUIREMENTS = [
    DisclosureRequirement(
        context=DisclosureContext.REGISTRATION,
        required=True,
        reason="User begins interaction with AI-powered platform",
        article_reference="Article 50(1)"
    ),
    DisclosureRequirement(
        context=DisclosureContext.LIVE_TRADING_ACTIVATION,
        required=True,
        reason="AI system will execute real trades",
        article_reference="Article 50(1)"
    ),
    DisclosureRequirement(
        context=DisclosureContext.API_RESPONSE,
        required=True,
        reason="Programmatic access to AI outputs",
        article_reference="Article 50(1), Recital 132"
    ),
]

class TransparencyDisclosureManager:
    """
    Manages Article 50 transparency disclosures.

    Responsibilities:
    1. Generate disclosure content
    2. Track user acknowledgments
    3. Provide audit trail
    4. API response headers
    """

    def __init__(self, storage_backend=None):
        self.storage = storage_backend or {}
        self.disclosures: dict[str, AIDisclosure] = {}

    def create_disclosure(
        self,
        user_id: str,
        context: DisclosureContext,
        disclosure_type: DisclosureType = DisclosureType.AI_INTERACTION
    ) -> AIDisclosure:
        """Create new disclosure for user context."""
        disclosure_id = self._generate_id(user_id, context)
        disclosure = AIDisclosure(
            disclosure_id=disclosure_id,
            disclosure_type=disclosure_type,
            context=context,
            timestamp=datetime.utcnow(),
            user_id=user_id
        )
        self.disclosures[disclosure_id] = disclosure
        return disclosure

    def record_acknowledgment(
        self,
        disclosure_id: str,
        user_id: str
    ) -> bool:
        """Record user acknowledgment of AI disclosure."""
        if disclosure_id not in self.disclosures:
            return False
        disclosure = self.disclosures[disclosure_id]
        if disclosure.user_id != user_id:
            return False
        disclosure.acknowledged = True
        disclosure.acknowledgment_timestamp = datetime.utcnow()
        return True

    def check_disclosure_required(
        self,
        user_id: str,
        context: DisclosureContext
    ) -> bool:
        """Check if disclosure is required for this context."""
        for req in DISCLOSURE_REQUIREMENTS:
            if req.context == context and req.required:
                # Check if already acknowledged
                disclosure_id = self._generate_id(user_id, context)
                if disclosure_id in self.disclosures:
                    return not self.disclosures[disclosure_id].acknowledged
                return True
        return False

    def get_api_headers(self) -> dict[str, str]:
        """Return HTTP headers for API responses per Article 50."""
        return {
            "X-AI-System": "true",
            "X-AI-Model": "Distributional-PPO-v1",
            "X-AI-Disclosure": "This response is generated by an AI system"
        }

    def _generate_id(self, user_id: str, context: DisclosureContext) -> str:
        """Generate unique disclosure ID."""
        data = f"{user_id}:{context.value}"
        return hashlib.sha256(data.encode()).hexdigest()[:16]

# Factory function
def create_transparency_manager() -> TransparencyDisclosureManager:
    """Create configured transparency manager."""
    return TransparencyDisclosureManager()
```

```python
"""
Article 53(1)(c) EU AI Act - Copyright Compliance for Training Data.

References:
- EU AI Act Article 53(1)(c)
- DSM Directive 2019/790 Article 4 (Text and Data Mining)
- GPAI Code of Practice Chapter on Copyright
"""
from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum
from typing import Optional
import hashlib

class DataSourceType(Enum):
    """Types of training data sources."""
    PUBLIC_MARKET_DATA = "public_market_data"
    LICENSED_DATA = "licensed_data"
    OPEN_DATA = "open_data"
    PROPRIETARY = "proprietary"
    SYNTHETIC = "synthetic"

class CopyrightStatus(Enum):
    """Copyright status of data source."""
    PUBLIC_DOMAIN = "public_domain"
    LICENSED = "licensed"
    FAIR_USE = "fair_use"
    TDM_EXCEPTION = "tdm_exception"  # Text and Data Mining exception
    OPT_OUT_RESPECTED = "opt_out_respected"
    NOT_APPLICABLE = "not_applicable"

@dataclass
class DataSourceRecord:
    """Record of a training data source for copyright compliance."""
    source_id: str
    source_name: str
    source_type: DataSourceType
    copyright_status: CopyrightStatus
    provider: str
    license_type: Optional[str] = None
    license_url: Optional[str] = None
    opt_out_checked: bool = False
    opt_out_check_date: Optional[datetime] = None
    opt_out_mechanism: Optional[str] = None  # robots.txt, TDMRep, etc.
    description: str = ""
    date_added: datetime = field(default_factory=datetime.utcnow)

@dataclass
class OptOutCheck:
    """Record of opt-out verification per Article 4(3) DSM Directive."""
    source_id: str
    check_date: datetime
    mechanism_checked: str  # robots.txt, TDMRep header, meta tag
    opt_out_found: bool
    action_taken: str  # "excluded", "not_applicable", "proceeded"
    evidence_hash: Optional[str] = None  # Hash of robots.txt etc.

class CopyrightComplianceManager:
    """
    Manages copyright compliance for AI training data.

    Per Article 53(1)(c) EU AI Act:
    "put in place a policy to comply with Union law on copyright
    and related rights, and in particular to identify and comply
    with, including through state-of-the-art technologies, a
    reservation of rights expressed pursuant to Article 4(3) of
    Directive (EU) 2019/790"
    """

    def __init__(self):
        self.data_sources: dict[str, DataSourceRecord] = {}
        self.opt_out_checks: list[OptOutCheck] = []
        self._initialize_default_sources()

    def _initialize_default_sources(self):
        """Initialize known training data sources."""
        # Market data sources (typically not copyrightable)
        sources = [
            DataSourceRecord(
                source_id="binance_ohlcv",
                source_name="Binance OHLCV Data",
                source_type=DataSourceType.PUBLIC_MARKET_DATA,
                copyright_status=CopyrightStatus.NOT_APPLICABLE,
                provider="Binance",
                description="Historical OHLCV price data via public API",
                opt_out_checked=True
            ),
            DataSourceRecord(
                source_id="polygon_stocks",
                source_name="Polygon.io Stock Data",
                source_type=DataSourceType.LICENSED_DATA,
                copyright_status=CopyrightStatus.LICENSED,
                provider="Polygon.io",
                license_type="Commercial API License",
                license_url="https://polygon.io/terms",
                description="US equity market data",
                opt_out_checked=True
            ),
        ]
        for source in sources:
            self.data_sources[source.source_id] = source

    def register_data_source(self, source: DataSourceRecord) -> str:
        """Register a new training data source."""
        self.data_sources[source.source_id] = source
        return source.source_id

    def check_opt_out(
        self,
        source_id: str,
        mechanism: str,
        content_hash: Optional[str] = None
    ) -> OptOutCheck:
        """
        Check and record opt-out status per Article 4(3) DSM Directive.

        Mechanisms:
        - robots.txt: ai-training-opt-out directive
        - TDMRep: HTTP header or meta tag
        - Manual declaration
        """
        check = OptOutCheck(
            source_id=source_id,
            check_date=datetime.utcnow(),
            mechanism_checked=mechanism,
            opt_out_found=False,  # Would be determined by actual check
            action_taken="not_applicable",
            evidence_hash=content_hash
        )
        self.opt_out_checks.append(check)

        if source_id in self.data_sources:
            self.data_sources[source_id].opt_out_checked = True
            self.data_sources[source_id].opt_out_check_date = check.check_date
            self.data_sources[source_id].opt_out_mechanism = mechanism

        return check

    def get_compliance_status(self) -> dict:
        """Get overall copyright compliance status."""
        total = len(self.data_sources)
        checked = sum(1 for s in self.data_sources.values() if s.opt_out_checked)
        licensed = sum(
            1 for s in self.data_sources.values()
            if s.copyright_status == CopyrightStatus.LICENSED
        )

        return {
            "total_sources": total,
            "opt_out_checked": checked,
            "licensed_sources": licensed,
            "compliance_percentage": (checked / total * 100) if total > 0 else 100,
            "last_audit": datetime.utcnow().isoformat()
        }

    def generate_policy_document(self) -> str:
        """Generate copyright policy document for public disclosure."""
        return COPYRIGHT_POLICY_TEMPLATE.format(
            date=datetime.utcnow().strftime("%Y-%m-%d"),
            sources_count=len(self.data_sources),
            compliance_status=self.get_compliance_status()
        )

    def get_training_data_sources(self) -> list[dict]:
        """Get list of training data sources for Article 53(1)(d) summary."""
        return [
            {
                "name": s.source_name,
                "type": s.source_type.value,
                "copyright_status": s.copyright_status.value,
                "provider": s.provider,
                "license": s.license_type
            }
            for s in self.data_sources.values()
        ]
```

### 2.3 User AI Acknowledgment

```python
"""
User AI Acknowledgment System.

Captures explicit user acknowledgment per Article 50 before
accessing AI-powered features.
"""
from dataclasses import dataclass
from datetime import datetime
from typing import Optional
from enum import Enum
import hashlib

class AcknowledgmentType(Enum):
    """Types of acknowledgments required."""
    AI_SYSTEM_AWARENESS = "ai_system_awareness"
    RISK_UNDERSTANDING = "risk_understanding"
    LIMITATION_ACCEPTANCE = "limitation_acceptance"
    LIVE_TRADING_CONSENT = "live_trading_consent"

@dataclass
class UserAcknowledgment:
    """Record of user acknowledgment."""
    acknowledgment_id: str
    user_id: str
    acknowledgment_type: AcknowledgmentType
    timestamp: datetime
    ip_address: Optional[str] = None
    user_agent: Optional[str] = None
    version: str = "1.0"

    # The actual text acknowledged
    text_hash: str = ""

    @classmethod
    def create(
        cls,
        user_id: str,
        ack_type: AcknowledgmentType,
        text_content: str,
        ip_address: Optional[str] = None,
        user_agent: Optional[str] = None
    ) -> "UserAcknowledgment":
        """Create new acknowledgment record."""
        ack_id = hashlib.sha256(
            f"{user_id}:{ack_type.value}:{datetime.utcnow().isoformat()}".encode()
        ).hexdigest()[:16]
        text_hash = hashlib.sha256(text_content.encode()).hexdigest()

        return cls(
            acknowledgment_id=ack_id,
            user_id=user_id,
            acknowledgment_type=ack_type,
            timestamp=datetime.utcnow(),
            ip_address=ip_address,
            user_agent=user_agent,
            text_hash=text_hash
        )

ACKNOWLEDGMENT_TEXTS = {
    AcknowledgmentType.AI_SYSTEM_AWARENESS: """
I acknowledge that:

1. This platform uses an Artificial Intelligence (AI) system based on
   machine learning technology (Reinforcement Learning).

2. Trading signals and predictions are generated by the AI model,
   not by human analysts.

3. I am interacting with an AI system when using the trading features
   of this platform.
""",
    AcknowledgmentType.RISK_UNDERSTANDING: """
I understand and accept that:

1. AI predictions may be inaccurate or suboptimal.
2. Past performance does not guarantee future results.
3. I may experience significant financial losses.
4. The AI system has limitations and may behave unexpectedly.
""",
    AcknowledgmentType.LIMITATION_ACCEPTANCE: """
I acknowledge the following AI system limitations:

1. The model was trained on historical data that may not reflect
   future market conditions.
2. Performance may degrade during unprecedented market events.
3. The system requires human oversight for safe operation.
4. I am responsible for monitoring AI outputs and making final decisions.
""",
    AcknowledgmentType.LIVE_TRADING_CONSENT: """
Before enabling live trading, I confirm that:

1. I understand I am using an AI-powered trading system.
2. I accept full responsibility for all trading decisions.
3. I will maintain appropriate oversight of the AI system.
4. I understand that substantial losses are possible.
5. I am trading with funds I can afford to lose.
"""
}

class UserAcknowledgmentManager:
    """Manages user acknowledgments for AI features."""

    def __init__(self):
        self.acknowledgments: dict[str, list[UserAcknowledgment]] = {}

    def get_required_acknowledgments(
        self,
        user_id: str,
        feature: str
    ) -> list[AcknowledgmentType]:
        """Get acknowledgments required for a feature."""
        feature_requirements = {
            "registration": [AcknowledgmentType.AI_SYSTEM_AWARENESS],
            "strategy_creation": [
                AcknowledgmentType.AI_SYSTEM_AWARENESS,
                AcknowledgmentType.LIMITATION_ACCEPTANCE
            ],
            "live_trading": [
                AcknowledgmentType.AI_SYSTEM_AWARENESS,
                AcknowledgmentType.RISK_UNDERSTANDING,
                AcknowledgmentType.LIMITATION_ACCEPTANCE,
                AcknowledgmentType.LIVE_TRADING_CONSENT
            ]
        }

        required = feature_requirements.get(feature, [])
        acknowledged = self._get_user_acknowledged_types(user_id)

        return [r for r in required if r not in acknowledged]

    def record_acknowledgment(
        self,
        user_id: str,
        ack_type: AcknowledgmentType,
        ip_address: Optional[str] = None,
        user_agent: Optional[str] = None
    ) -> UserAcknowledgment:
        """Record user acknowledgment."""
        text = ACKNOWLEDGMENT_TEXTS.get(ack_type, "")
        ack = UserAcknowledgment.create(
            user_id=user_id,
            ack_type=ack_type,
            text_content=text,
            ip_address=ip_address,
            user_agent=user_agent
        )

        if user_id not in self.acknowledgments:
            self.acknowledgments[user_id] = []
        self.acknowledgments[user_id].append(ack)

        return ack

    def check_feature_access(
        self,
        user_id: str,
        feature: str
    ) -> tuple[bool, list[AcknowledgmentType]]:
        """Check if user can access feature (has all acknowledgments)."""
        missing = self.get_required_acknowledgments(user_id, feature)
        return (len(missing) == 0, missing)

    def _get_user_acknowledged_types(
        self,
        user_id: str
    ) -> set[AcknowledgmentType]:
        """Get types user has already acknowledged."""
        if user_id not in self.acknowledgments:
            return set()
        return {a.acknowledgment_type for a in self.acknowledgments[user_id]}

    def get_acknowledgment_text(
        self,
        ack_type: AcknowledgmentType
    ) -> str:
        """Get acknowledgment text for display."""
        return ACKNOWLEDGMENT_TEXTS.get(ack_type, "")

def create_acknowledgment_manager() -> UserAcknowledgmentManager:
    """Factory function."""
    return UserAcknowledgmentManager()
```
